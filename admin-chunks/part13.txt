    $('[data-upcoming-jobs]').innerHTML=jobs.filter(job=>new Date(job.scheduled_start)>=new Date()&&!['completed','cancelled'].includes(job.status)).slice(0,6).map(job=>`<article class="schedule-item"><div class="schedule-time"><span>${new Date(job.scheduled_start).toLocaleDateString([],{month:'short',day:'numeric'})}</span><small>${new Date(job.scheduled_start).toLocaleTimeString([],{hour:'numeric',minute:'2-digit'})}</small></div><div><strong>${esc(job.service_name)}</strong><p>${esc(job.customer_name)} · ${esc(job.address)}</p></div>${badge(job.status)}</article>`).join('')||'<div class="empty-admin">No upcoming jobs.</div>';
  }
