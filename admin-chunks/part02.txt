  async function requireSession(){
    if(!client){location.href='admin-login.html';return false}
    const {data:{session}}=await client.auth.getSession();
    if(!session){location.href='admin-login.html';return false}
    const {data:profile,error}=await client.from('svc_profiles').select('*').eq('id',session.user.id).eq('active',true).maybeSingle();
    if(error||!profile){await client.auth.signOut();location.href='admin-login.html';return false}
    state.session=session;state.profile=profile;applyRole();return true;
  }

  function applyRole(){
    const role=state.profile.role;
    $$('[data-roles]').forEach(el=>{const allowed=el.dataset.roles.split(',');el.hidden=!allowed.includes(role)});
    $('[data-user-name]').textContent=state.profile.full_name||state.profile.email;
    $('[data-user-role]').textContent=role;
    $('[data-admin-role]').textContent=`${role} account`;
  }

  function setView(name){
    const button=$(`[data-view-button="${name}"]`);if(button?.hidden)return;
    $$('[data-view]').forEach(view=>view.classList.toggle('active',view.dataset.view===name));
    $$('[data-view-button]').forEach(btn=>btn.classList.toggle('active',btn.dataset.viewButton===name));
    $('[data-view-title]').textContent=button?.textContent.trim().replace(/\d+$/,'')||'Admin';
    $('.admin-sidebar')?.classList.remove('open');window.scrollTo({top:0,behavior:'smooth'});
  }

  }
