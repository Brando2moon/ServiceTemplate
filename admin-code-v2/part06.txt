  function projectStoragePath(url=''){
    const marker='/storage/v1/object/public/service-public/';
    const index=String(url).indexOf(marker);
    return index<0?null:decodeURIComponent(String(url).slice(index+marker.length));
  }

  function updateProjectPhotoPreview(kind,url=''){
    const image=$(`[data-${kind}-photo-preview]`);
    const empty=$(`[data-${kind}-photo-empty]`);

  }

  function validateProjectPhoto(file){
    if(!file)return;
    if(!['image/jpeg','image/png','image/webp'].includes(file.type))throw new Error('Photos must be JPG, PNG, or WebP.');
    if(file.size>8*1024*1024)throw new Error('Each photo must be 8 MB or smaller.');
  }
