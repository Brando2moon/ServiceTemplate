  async function removeProjectPhoto(kind){
    const form=$('[data-project-form]');
    const id=form.elements.id.value;
    const field=`${kind}_image_url`;
    const currentUrl=form.elements[field].value.trim();
    if(!currentUrl)return;
    if(!confirm(`Remove the ${kind} photo? The project will be unpublished until both photos are added.`))return;
    const status=$('[data-project-status]');
    try{
      status.textContent=`Removing ${kind} photo…`;
      if(id){
        const {error}=await client.from('svc_projects').update({[field]:null,published:false,updated_at:new Date().toISOString()}).eq('id',id);
        if(error)throw error;
      }
      await deleteStoredProjectPhoto(currentUrl);
      form.elements[field].value='';
      form.elements.published.checked=false;
      updateProjectPhotoPreview(kind,'');
      status.textContent=`${kind[0].toUpperCase()+kind.slice(1)} photo removed.`;
      await loadProjects();
    }catch(error){console.error(error);status.textContent=error.message||'The photo could not be removed.'}
  }

  async function deleteProject(id){
    const item=state.projects.find(row=>row.id===id);
    if(!item||!confirm('Delete this project and its uploaded photos?'))return;
    const {error}=await client.from('svc_projects').delete().eq('id',id);
    if(error){showAlert(error.message,true);return}
    await Promise.allSettled([deleteStoredProjectPhoto(item.before_image_url),deleteStoredProjectPhoto(item.after_image_url)]);
    showAlert('Project deleted.');
    clearProjectForm();
    await loadProjects();
  }
