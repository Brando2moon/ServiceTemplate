  async function loadJobs(){
    const {data,error}=await client.from('svc_jobs').select('*,assigned:svc_profiles!svc_jobs_assigned_to_fkey(full_name,email,role)').order('scheduled_start',{ascending:true});
    if(error){showAlert('Unable to load jobs.',true);return}
    state.jobs=data||[];renderJobs();
  }

  function jobsForRange(){
    let jobs=[...state.jobs];const filter=$('[data-job-date-filter]')?.value;if(filter)jobs=jobs.filter(job=>String(job.scheduled_start||'').slice(0,10)===filter);
    const now=new Date();let end=null;if(state.calendarRange==='today'){end=new Date(now);end.setHours(23,59,59,999)}if(state.calendarRange==='week')end=new Date(now.getTime()+7*86400000);if(state.calendarRange==='month')end=new Date(now.getTime()+30*86400000);
    if(end)jobs=jobs.filter(job=>new Date(job.scheduled_start)>=new Date(now.toDateString())&&new Date(job.scheduled_start)<=end);
    return jobs;
  }

  function renderJobs(){
    const board=$('[data-job-board]');if(!board)return;const jobs=jobsForRange();
    board.innerHTML=jobs.length?jobs.map(job=>`<article class="job-card ${state.currentJob?.id===job.id?'selected':''}" data-open-job="${job.id}"><time>${fmt(job.scheduled_start)}</time><h3>${esc(job.service_name)}</h3><p>${esc(job.customer_name)} · ${esc(job.address)}</p><div class="job-card-footer">${badge(job.status)}<small>${esc(job.assigned?.full_name||'Unassigned')}</small></div></article>`).join(''):'<div class="empty-admin">No jobs in this calendar range.</div>';
  }
